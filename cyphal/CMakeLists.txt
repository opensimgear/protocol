cmake_minimum_required(VERSION 3.18)
if(CMAKE_CURRENT_SOURCE_DIR STREQUAL CMAKE_SOURCE_DIR)
  project(opensimgear_cyphal C)
endif()
add_library(osg_cyphal STATIC src/gateway_protocol.c src/cyphal.c vendor/libcanard/canard.c)
target_include_directories(osg_cyphal PUBLIC include generated vendor/libcanard)
target_compile_features(osg_cyphal PUBLIC c_std_11)
set_target_properties(osg_cyphal PROPERTIES POSITION_INDEPENDENT_CODE ON)
option(OSG_CYPHAL_BUILD_TESTS "Build portable Cyphal transport tests" OFF)
if(OSG_CYPHAL_BUILD_TESTS)
  enable_testing()
  add_executable(osg_cyphal_test tests/transport_test.c)
  target_link_libraries(osg_cyphal_test PRIVATE osg_cyphal)
  target_compile_options(osg_cyphal_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  add_test(NAME osg_cyphal_transport COMMAND osg_cyphal_test)
endif()
